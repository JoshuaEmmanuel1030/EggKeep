-- ============================================================================
-- PART 0 — ORPHAN DETECTION (READ-ONLY)
-- Names in transactional tables that do NOT resolve to a live item_types row.
-- "deleted_match" = TRUE means the name exists in item_types but is
-- soft-deleted (fix by restoring the catalog row, usually).
-- ============================================================================

-- 0.1 inflows
SELECT f.category, f.product,
       count(*)                                   AS row_count,
       count(*) FILTER (WHERE f.voided_at IS NULL) AS live_rows,
       sum(f.remaining_butir) FILTER (WHERE f.voided_at IS NULL) AS live_remaining_stock,
       bool_or(itd.id IS NOT NULL)                AS deleted_match
FROM inflows f
LEFT JOIN item_types it
       ON it.category = f.category AND it.name = f.product AND it.deleted_at IS NULL
LEFT JOIN item_types itd
       ON itd.category = f.category AND itd.name = f.product AND itd.deleted_at IS NOT NULL
WHERE it.id IS NULL
GROUP BY f.category, f.product
ORDER BY live_remaining_stock DESC NULLS LAST, row_count DESC;

-- 0.2 outflows
SELECT o.category, o.product,
       count(*)                                   AS row_count,
       count(*) FILTER (WHERE o.voided_at IS NULL) AS live_rows,
       bool_or(itd.id IS NOT NULL)                AS deleted_match
FROM outflows o
LEFT JOIN item_types it
       ON it.category = o.category AND it.name = o.product AND it.deleted_at IS NULL
LEFT JOIN item_types itd
       ON itd.category = o.category AND itd.name = o.product AND itd.deleted_at IS NOT NULL
WHERE it.id IS NULL
GROUP BY o.category, o.product
ORDER BY row_count DESC;

-- 0.3 pack_skus.egg_product  (this is the query that catches the
--     'BEBEK ASIN MATANG' vs 'ASIN MATENG' incident class)
SELECT s.code, s.display_name, s.egg_product,
       (s.deleted_at IS NOT NULL) AS sku_soft_deleted,
       bool_or(itd.id IS NOT NULL) AS deleted_match
FROM pack_skus s
LEFT JOIN item_types it
       ON it.category = 'egg' AND it.name = s.egg_product AND it.deleted_at IS NULL
LEFT JOIN item_types itd
       ON itd.category = 'egg' AND itd.name = s.egg_product AND itd.deleted_at IS NOT NULL
WHERE it.id IS NULL
GROUP BY s.code, s.display_name, s.egg_product, s.deleted_at;

-- 0.4 pack_skus.packaging_item (NULL packaging is legitimate — excluded)
SELECT s.code, s.display_name, s.packaging_item,
       (s.deleted_at IS NOT NULL) AS sku_soft_deleted,
       bool_or(itd.id IS NOT NULL) AS deleted_match
FROM pack_skus s
LEFT JOIN item_types it
       ON it.category = 'packaging' AND it.name = s.packaging_item AND it.deleted_at IS NULL
LEFT JOIN item_types itd
       ON itd.category = 'packaging' AND itd.name = s.packaging_item AND itd.deleted_at IS NOT NULL
WHERE s.packaging_item IS NOT NULL
  AND it.id IS NULL
GROUP BY s.code, s.display_name, s.packaging_item, s.deleted_at;

-- 0.5 Cross-category duplicate names (relevant to the recalculate_inventory_fifo
--     category-matching fix — see PLAN.md open question 5)
SELECT name, array_agg(DISTINCT category) AS categories, count(*)
FROM item_types
GROUP BY name
HAVING count(DISTINCT category) > 1;

-- ============================================================================
-- PART 2 — BACKFILL (Stage A)
-- Joins include soft-deleted item_types rows (historical stock of a
-- since-deleted item still deserves its UUID), but prefers the LIVE row when
-- both a live and a deleted row share (category, name).
-- ============================================================================

WITH resolved AS (
  SELECT DISTINCT ON (category, name) id, category, name
  FROM item_types
  ORDER BY category, name, (deleted_at IS NULL) DESC, created_at ASC
)
UPDATE public.inflows f
SET item_type_id = r.id
FROM resolved r
WHERE f.item_type_id IS NULL
  AND r.category = f.category
  AND r.name = f.product;

WITH resolved AS (
  SELECT DISTINCT ON (category, name) id, category, name
  FROM item_types
  ORDER BY category, name, (deleted_at IS NULL) DESC, created_at ASC
)
UPDATE public.outflows o
SET item_type_id = r.id
FROM resolved r
WHERE o.item_type_id IS NULL
  AND r.category = o.category
  AND r.name = o.product;

WITH resolved AS (
  SELECT DISTINCT ON (name) id, name
  FROM item_types
  WHERE category = 'egg'
  ORDER BY name, (deleted_at IS NULL) DESC, created_at ASC
)
UPDATE public.pack_skus s
SET egg_item_type_id = r.id
FROM resolved r
WHERE s.egg_item_type_id IS NULL
  AND r.name = s.egg_product;

WITH resolved AS (
  SELECT DISTINCT ON (name) id, name
  FROM item_types
  WHERE category = 'packaging'
  ORDER BY name, (deleted_at IS NULL) DESC, created_at ASC
)
UPDATE public.pack_skus s
SET packaging_item_type_id = r.id
FROM resolved r
WHERE s.packaging_item_type_id IS NULL
  AND s.packaging_item IS NOT NULL
  AND r.name = s.packaging_item;

-- ============================================================================
-- PART 4 — RESOLVER TRIGGERS (Stage A) — the stale-client defense.
-- Any client (old bundle, offline sync replay, manual SQL) that inserts a
-- name-only row gets its item_type_id filled server-side. Resolution prefers
-- live catalog rows, falls back to soft-deleted ones, and leaves NULL (does
-- NOT raise) if nothing matches — enforcement is Stage D's job.
-- ============================================================================

CREATE OR REPLACE FUNCTION public.resolve_item_type_id()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  IF NEW.item_type_id IS NULL AND NEW.product IS NOT NULL THEN
    SELECT id INTO NEW.item_type_id
    FROM item_types
    WHERE category = NEW.category
      AND name = NEW.product
    ORDER BY (deleted_at IS NULL) DESC, created_at ASC
    LIMIT 1;
  END IF;
  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS trg_inflows_resolve_item_type ON public.inflows;
CREATE TRIGGER trg_inflows_resolve_item_type
  BEFORE INSERT ON public.inflows
  FOR EACH ROW EXECUTE FUNCTION public.resolve_item_type_id();

DROP TRIGGER IF EXISTS trg_outflows_resolve_item_type ON public.outflows;
CREATE TRIGGER trg_outflows_resolve_item_type
  BEFORE INSERT ON public.outflows
  FOR EACH ROW EXECUTE FUNCTION public.resolve_item_type_id();

-- pack_skus resolver (two columns, so its own function)
CREATE OR REPLACE FUNCTION public.resolve_pack_sku_item_types()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  IF NEW.egg_item_type_id IS NULL AND NEW.egg_product IS NOT NULL THEN
    SELECT id INTO NEW.egg_item_type_id
    FROM item_types
    WHERE category = 'egg' AND name = NEW.egg_product
    ORDER BY (deleted_at IS NULL) DESC, created_at ASC
    LIMIT 1;
  END IF;
  IF NEW.packaging_item_type_id IS NULL AND NEW.packaging_item IS NOT NULL THEN
    SELECT id INTO NEW.packaging_item_type_id
    FROM item_types
    WHERE category = 'packaging' AND name = NEW.packaging_item
    ORDER BY (deleted_at IS NULL) DESC, created_at ASC
    LIMIT 1;
  END IF;
  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS trg_pack_skus_resolve_item_types ON public.pack_skus;
CREATE TRIGGER trg_pack_skus_resolve_item_types
  BEFORE INSERT OR UPDATE OF egg_product, packaging_item ON public.pack_skus
  FOR EACH ROW EXECUTE FUNCTION public.resolve_pack_sku_item_types();

-- ---------------------------------------------------------------------------
-- record_order_outflows v2 — matches inventory by item_type_id.
-- Changes vs v1 (supabase/migrations/20260703000000_atomic_order_outflows.sql):
--   * resolves p_entry.item_type_id, falling back to (category, name) lookup
--   * RAISES 'UNKNOWN_PRODUCT' when nothing resolves (fail LOUD — this is the
--     error the BEBEK ASIN MATANG incident would have produced immediately)
--   * FIFO scan matches inflows by item_type_id, with a name fallback branch
--     for legacy rows whose item_type_id is NULL
--   * outflow row stores BOTH item_type_id and the catalog display name at
--     time of sale (name doubles as the immutable snapshot)
-- Unchanged: FOR UPDATE row-locking, all-or-nothing semantics, kg-native
-- quantity_butir handling (quantities are passed through untouched).
-- ---------------------------------------------------------------------------

CREATE OR REPLACE FUNCTION public.record_order_outflows(p_entries jsonb)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_user uuid := auth.uid();
  v_entry jsonb;
  v_outflow_id uuid;
  v_qty numeric;
  v_remaining numeric;
  v_deduct numeric;
  v_product text;
  v_category inventory_category;
  v_item_type_id uuid;
  v_catalog_name text;
  inflow_rec RECORD;
BEGIN
  IF v_user IS NULL THEN
    RAISE EXCEPTION 'AUTH_REQUIRED: must be signed in to record outflows';
  END IF;

  IF p_entries IS NULL OR jsonb_typeof(p_entries) <> 'array' OR jsonb_array_length(p_entries) = 0 THEN
    RAISE EXCEPTION 'NO_ENTRIES: p_entries must be a non-empty json array';
  END IF;

  FOR v_entry IN SELECT * FROM jsonb_array_elements(p_entries) LOOP
    v_product      := v_entry->>'product';
    v_category     := (v_entry->>'category')::inventory_category;
    v_qty          := (v_entry->>'quantity_butir')::numeric;
    v_item_type_id := NULLIF(v_entry->>'item_type_id', '')::uuid;

    IF v_qty IS NULL OR v_qty <= 0 THEN
      RAISE EXCEPTION 'INVALID_ENTRY: a positive quantity_butir is required (got % / %)',
        v_product, v_qty;
    END IF;

    -- Resolve the catalog item: explicit id from a dual-write client, else
    -- server-side (category, name) lookup for old bundles. Prefer live rows.
    IF v_item_type_id IS NULL THEN
      IF v_product IS NULL THEN
        RAISE EXCEPTION 'INVALID_ENTRY: item_type_id or product name is required';
      END IF;
      SELECT id INTO v_item_type_id
      FROM item_types
      WHERE category = v_category AND name = v_product
      ORDER BY (deleted_at IS NULL) DESC, created_at ASC
      LIMIT 1;
      IF v_item_type_id IS NULL THEN
        RAISE EXCEPTION 'UNKNOWN_PRODUCT: % (%) is not in the catalog — check the SKU/catalog spelling',
          v_product, v_category;
      END IF;
    END IF;

    -- Canonical display name at time of sale (also our name snapshot).
    SELECT name INTO v_catalog_name FROM item_types WHERE id = v_item_type_id;
    IF v_catalog_name IS NULL THEN
      RAISE EXCEPTION 'UNKNOWN_PRODUCT: item_type_id % does not exist', v_item_type_id;
    END IF;

    INSERT INTO outflows (id, date, product, item_type_id, quantity_butir, user_id, category, invoice_supplier)
    VALUES (
      COALESCE(NULLIF(v_entry->>'id', '')::uuid, gen_random_uuid()),
      v_entry->>'date',
      v_catalog_name,
      v_item_type_id,
      v_qty,
      v_user,
      v_category,
      NULLIF(v_entry->>'invoice_supplier', '')
    )
    RETURNING id INTO v_outflow_id;

    v_remaining := v_qty;

    -- FIFO over live batches, matched by item_type_id. The name+category
    -- fallback branch covers legacy inflow rows that never got backfilled
    -- (deliberate Stage-0 orphans). FOR UPDATE semantics unchanged from v1.
    FOR inflow_rec IN
      SELECT id, remaining_butir
      FROM inflows
      WHERE (
              item_type_id = v_item_type_id
              OR (item_type_id IS NULL AND product = v_catalog_name AND category = v_category)
            )
        AND voided_at IS NULL
        AND remaining_butir > 0
      ORDER BY date ASC, created_at ASC
      FOR UPDATE
    LOOP
      EXIT WHEN v_remaining <= 0;

      v_deduct := LEAST(inflow_rec.remaining_butir, v_remaining);

      UPDATE inflows
      SET remaining_butir = remaining_butir - v_deduct
      WHERE id = inflow_rec.id;

      INSERT INTO fifo_deductions (outflow_id, inflow_id, quantity_deducted)
      VALUES (v_outflow_id, inflow_rec.id, v_deduct);

      v_remaining := v_remaining - v_deduct;
    END LOOP;

    IF v_remaining > 0 THEN
      RAISE EXCEPTION 'INSUFFICIENT_STOCK: % (%) is short % butir/pcs',
        v_catalog_name, v_category, v_remaining;
    END IF;
  END LOOP;
END;
$$;

GRANT EXECUTE ON FUNCTION public.record_order_outflows(jsonb) TO authenticated;

-- ---------------------------------------------------------------------------
-- recalculate_inventory_fifo v2 — id-based matching.
-- Changes vs v1 (supabase/migrations/20260122153642_...sql):
--   * inflow matching keys on outflow.item_type_id when present
--   * fallback for legacy rows matches product AND category (v1 matched
--     product only — cross-category leak fixed; see PLAN.md open question 5)
--   * summary groups by item_type_id (rename no longer splits a product's
--     rebuild stats); product_name reported via catalog join with fallback
--     to the stored snapshot. Return shape unchanged — no client change
--     needed in src/hooks/useRecalculateInventory.ts.
-- ---------------------------------------------------------------------------

CREATE OR REPLACE FUNCTION public.recalculate_inventory_fifo()
RETURNS TABLE(
  product_name TEXT,
  outflows_processed BIGINT,
  deductions_created BIGINT,
  total_deducted NUMERIC
)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  outflow_record RECORD;
  inflow_record RECORD;
  remaining_to_deduct NUMERIC;
  deduct_amount NUMERIC;
BEGIN
  DELETE FROM fifo_deductions WHERE true;

  UPDATE inflows
  SET remaining_butir = quantity_butir
  WHERE voided_at IS NULL;

  FOR outflow_record IN (
    SELECT id, product, category, item_type_id, quantity_butir, date, created_at
    FROM outflows
    WHERE voided_at IS NULL
    ORDER BY date ASC, created_at ASC
  ) LOOP
    remaining_to_deduct := outflow_record.quantity_butir;

    FOR inflow_record IN (
      SELECT id, remaining_butir
      FROM inflows
      WHERE (
              (outflow_record.item_type_id IS NOT NULL AND item_type_id = outflow_record.item_type_id)
              OR (
                   (outflow_record.item_type_id IS NULL OR item_type_id IS NULL)
                   AND product = outflow_record.product
                   AND category = outflow_record.category
                 )
            )
        AND voided_at IS NULL
        AND remaining_butir > 0
      ORDER BY date ASC, created_at ASC
    ) LOOP
      EXIT WHEN remaining_to_deduct <= 0;

      deduct_amount := LEAST(inflow_record.remaining_butir, remaining_to_deduct);

      INSERT INTO fifo_deductions (outflow_id, inflow_id, quantity_deducted)
      VALUES (outflow_record.id, inflow_record.id, deduct_amount);

      UPDATE inflows
      SET remaining_butir = remaining_butir - deduct_amount
      WHERE id = inflow_record.id;

      remaining_to_deduct := remaining_to_deduct - deduct_amount;
    END LOOP;
  END LOOP;

  RETURN QUERY
  SELECT
    COALESCE(it.name, o.product) AS product_name,
    COUNT(DISTINCT o.id) AS outflows_processed,
    COUNT(fd.id) AS deductions_created,
    COALESCE(SUM(fd.quantity_deducted), 0) AS total_deducted
  FROM outflows o
  LEFT JOIN item_types it ON it.id = o.item_type_id
  LEFT JOIN fifo_deductions fd ON fd.outflow_id = o.id
  WHERE o.voided_at IS NULL
  GROUP BY COALESCE(it.name, o.product)
  ORDER BY COALESCE(it.name, o.product);
END;
$$;
